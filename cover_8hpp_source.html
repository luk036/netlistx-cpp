49template <typename ViolateFunc, typename WeightMap, typename SolutionSet>
50auto pd_cover(ViolateFunc violate, WeightMap& weight, SolutionSet& soln)
51 -> std::pair<SolutionSet, typename WeightMap::mapped_type> {
52 using CostType = typename WeightMap::mapped_type;
53 using NodeType = typename SolutionSet::value_type;
54
55 CostType total_dual_cost = 0;
56 auto gap = weight; // copy weights
57 std::vector<NodeType> added_order;
58
59 // Phase 1: Primal-Dual Selection
60 for (auto&& violate_set : violate()) {
61 if (violate_set.empty()) continue;
62
63 auto min_vtx
64 = *std::min_element(violate_set.begin(), violate_set.end(),
65 [&](const auto& v1, const auto& v2) { return gap[v1] < gap[v2]; });
66 auto min_val = gap[min_vtx];
67
68 if (!soln.contains(min_vtx)) {
69 soln.insert(min_vtx);
70 added_order.emplace_back(min_vtx);
71 }
72
73 total_dual_cost += min_val;
74
75 for (const auto& vtx : violate_set) {
76 gap[vtx] -= min_val;
77 }
78 }
79
80 // Phase 2: Reverse-Delete Post-Processing
81 for (auto it = added_order.rbegin(); it != added_order.rend(); ++it) {
82 soln.erase(*it);
83 bool is_redundant = true;
84 for (auto&& check_set : violate()) {
85 if (!check_set.empty()) {
86 is_redundant = false;
87 break;
88 }
89 }
90 if (!is_redundant) {
91 soln.insert(*it);
92 }
93 }
94
95 CostType final_prml_cost = 0;
96 for (const auto& vtx : soln) {
97 final_prml_cost += weight[vtx];
98 }
99
100 assert(total_dual_cost <= final_prml_cost);
101 return std::make_pair(soln, final_prml_cost);
102}
