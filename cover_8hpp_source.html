50template <typename ViolateFunc, typename WeightMap, typename SolutionSet>
51auto pd_cover(ViolateFunc violate, WeightMap& weight, SolutionSet& soln)
52 -> std::pair<SolutionSet, typename WeightMap::mapped_type> {
53 using CostType = typename WeightMap::mapped_type;
54 using NodeType = typename SolutionSet::value_type;
55
56 CostType total_dual_cost = 0;
57 auto gap = weight; // copy weights
58 std::vector<NodeType> added_order;
59
60 // Phase 1: Primal-Dual Selection
61 for (auto&& violate_set : violate()) {
62 if (violate_set.empty()) continue;
63
64 auto min_vtx
65 = *std::min_element(violate_set.begin(), violate_set.end(),
66 [&](const auto& v1, const auto& v2) { return gap[v1] < gap[v2]; });
67 auto min_val = gap[min_vtx];
68
69 if (!soln.contains(min_vtx)) {
70 soln.insert(min_vtx);
71 added_order.emplace_back(min_vtx);
72 }
73
74 total_dual_cost += min_val;
75
76 for (const auto& vtx : violate_set) {
77 gap[vtx] -= min_val;
78 }
79 }
80
81 // Phase 2: Reverse-Delete Post-Processing
82 auto is_valid = [&]() -> bool {
83 for (auto&& check_set : violate()) {
84 if (!check_set.empty()) {
85 return false;
86 }
87 }
88 return true;
89 };
90 netlistx::detail::reverse_delete(soln, added_order, is_valid);
91
92 CostType final_prml_cost = 0;
93 for (const auto& vtx : soln) {
94 final_prml_cost += weight[vtx];
95 }
96
97 assert(total_dual_cost <= final_prml_cost);
98 return std::make_pair(soln, final_prml_cost);
99}
