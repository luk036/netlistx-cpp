46template <typename Gnl, typename C1, typename C2>
47auto min_vertex_cover(const Gnl& hyprgraph, const C1& weight, C2& coverset) ->
48 typename C1::mapped_type {
49 using T = typename C1::mapped_type;
50 auto in_coverset = [&](const auto& v) { return coverset.contains(v); };
51 [[maybe_unused]] auto total_dual_cost = T(0);
52 auto total_primal_cost = T(0);
53 auto gap = weight;
54 for (const auto& net : hyprgraph.nets) {
55 if (std::any_of(hyprgraph.gr[net].begin(), hyprgraph.gr[net].end(), in_coverset)) {
56 continue;
57 }
58
59 auto min_vtx
60 = *std::min_element(hyprgraph.gr[net].begin(), hyprgraph.gr[net].end(),
61 [&](const auto& v1, const auto& v2) { return gap[v1] < gap[v2]; });
62 auto min_val = gap[min_vtx];
63 coverset.insert(min_vtx);
64 total_primal_cost += weight[min_vtx];
65 total_dual_cost += min_val;
66 for (const auto& u : hyprgraph.gr[net]) {
67 gap[u] -= min_val;
68 }
69 }
70
71 assert(total_dual_cost <= total_primal_cost);
72 return total_primal_cost;
73}
