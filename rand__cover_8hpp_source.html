69template <typename Hypergraph, typename WeightMap, typename RNG>
70auto rand_hyper_vertex_cover_trial(const Hypergraph& hyprgraph, const WeightMap& weight,
71 const py::set<typename Hypergraph::node_t>& coverset, RNG& rng)
72 -> std::pair<py::set<typename Hypergraph::node_t>, typename WeightMap::mapped_type> {
73 using node_t = typename Hypergraph::node_t;
74 using CostType = typename WeightMap::mapped_type;
75
76 py::set<node_t> soln = coverset.copy();
77 std::vector<node_t> added_order;
78 std::uniform_real_distribution<double> dist(0.0, 1.0);
79
80 for (const auto& net : hyprgraph.nets) {
81 // Skip nets that are empty or already covered.
82 if (hyprgraph.gr[net].empty() || netlistx::detail::net_is_covered(hyprgraph, net, soln)) {
83 continue;
84 }
85
86 // Copy vertices to a vector for random access
87 std::vector<node_t> vertices(hyprgraph.gr[net].begin(), hyprgraph.gr[net].end());
88
89 // Generalized Pitt rule: P(pick vi) ~ 1/w(vi)
90 double total_inv{};
91 std::vector<double> inv_weights;
92 inv_weights.reserve(vertices.size());
93 for (const auto& v : vertices) {
94 const double inv = 1.0 / static_cast<double>(weight[v]);
95 inv_weights.push_back(inv);
96 total_inv += inv;
97 }
98
99 const double r = dist(rng);
100 double cumulative{};
101 node_t chosen{};
102 for (std::size_t i = 0; i < vertices.size(); ++i) {
103 cumulative += inv_weights[i] / total_inv;
104 if (r < cumulative) {
105 chosen = vertices[i];
106 break;
107 }
108 }
109
110 soln.insert(chosen);
111 added_order.push_back(chosen);
112 }
113
114 // Phase 2: Reverse-Delete Post-Processing
115 netlistx::detail::reverse_delete(
116 soln, added_order, [&]() { return netlistx::detail::all_nets_covered(hyprgraph, soln); });
117
118 CostType total_cost{};
119 for (const auto& v : soln) {
120 total_cost += weight[v];
121 }
122 return {std::move(soln), total_cost};
123}
